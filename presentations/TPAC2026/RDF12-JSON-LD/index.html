RDF Annotation

<Alice> :bought <LennyTheLion> {|
    a :Purchase ;
    :seller <ToyStore> ;
    :date "2026-10"
  |} .

Plain assertion

<Alice> :bought <LennyTheLion> .
{
  "@id": "Alice",
  "bought": {
    "@id": "LennyTheLion"
  }
}

Annotation Sugar

<Alice> :bought <LennyTheLion> {|
    a :Purchase ;
    :seller <ToyStore> ;
    :date "2026-10"
  |} .
{
  "@id": "Alice",
  "bought": {
    "@id": "LennyTheLion",
    "@annotation": {
      "@type": "Purchase",
      "seller": {"@id": "ToyStore"},
      "date": "2026-10"
    }
  }
}
